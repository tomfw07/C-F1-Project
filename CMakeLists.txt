cmake_minimum_required(VERSION 3.16)
project(F1TelemetryAnalyzer VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(f1_telemetry src/main.cpp)
target_include_directories(f1_telemetry PRIVATE include)
if(MSVC)
  target_compile_options(f1_telemetry PRIVATE /W4)
else()
  target_compile_options(f1_telemetry PRIVATE -Wall -Wextra -Wpedantic)
endif()
